const canvas = document.getElementById('scene'); const renderer = new THREE.WebGLRenderer({canvas, antialias:true}); renderer.setSize(innerWidth, innerHeight); renderer.setPixelRatio(Math.min(devicePixelRatio, 2));

const scene = new THREE.Scene(); const camera = new THREE.PerspectiveCamera(45, innerWidth/innerHeight, 0.1, 1000); camera.position.set(0, 0, 6);

const controls = new OrbitControls(camera, canvas); controls.enableDamping = true; controls.dampingFactor = 0.05; controls.minDistance = 3.2; controls.maxDistance = 14; controls.autoRotate = false;

// Lumière du soleil const sun = new THREE.DirectionalLight(0xffffff, 2.2); sun.position.set(5, 3, 5); scene.add(sun); scene.add(new THREE.AmbientLight(0x335577, 0.6));

// Étoiles de fond function makeStars(count, size){ const geo = new THREE.BufferGeometry(); const pos = new Float32Array(count*3); for(let i=0;i<count*3;i++) pos[i]=(Math.random()-0.5)*200; geo.setAttribute('position', new THREE.BufferAttribute(pos,3)); const mat = new THREE.PointsMaterial({color:0xffffff, size, sizeAttenuation:true, transparent:true, opacity:.9}); return new THREE.Points(geo, mat); } scene.add(makeStars(1500, 0.4));

const TX = 'https://cdn.jsdelivr.net/npm/three-globe/example/img/'; const loader = new THREE.TextureLoader(); loader.crossOrigin = 'anonymous';

const earthGroup = new THREE.Group(); scene.add(earthGroup);

const earthMat = new THREE.MeshPhongMaterial({ map: loader.load(TX+'earth-blue-marble.jpg'), specularMap: loader.load(TX+'earth-water.png'), specular: new THREE.Color(0x333333), shininess: 15 }); const earth = new THREE.Mesh(new THREE.SphereGeometry(2, 96, 96), earthMat); earthGroup.add(earth);

// Atmosphere const atmo = new THREE.Mesh( new THREE.SphereGeometry(2.12, 96, 96), new THREE.MeshBasicMaterial({color:0x3da9fc, transparent:true, opacity:.18, side:THREE.BackSide}) ); earthGroup.add(atmo);

// Nuages const cloudMat = new THREE.MeshPhongMaterial({map:loader.load(TX+'earth-topology.png'), transparent:true, opacity:.4}); const clouds = new THREE.Mesh(new THREE.SphereGeometry(2.04, 96, 96), cloudMat); earthGroup.add(clouds);

// Fin du chargement let loaded=0; const total=4; function check(){ if(++loaded>=total){ document.getElementById('loader').style.display='none'; } } [earthMat.map, earthMat.specularMap, cloudMat.map].forEach(t=>{ t? t.onLoad=check : check(); }); setTimeout(()=>document.getElementById('loader').style.display='none', 4000);
